### docker-compose.yaml文件配置

```
version: '3.8'

services:
  # ========== Master 节点 ==========
  master:
    image: centos:7
    container_name: cluster-master
    hostname: master
    stdin_open: true        # 保持 STDIN 打开 (类似 -i)
    tty: true               # 分配伪 TTY (类似 -t)
    restart: unless-stopped
    networks:
      cluster-net:
        ipv4_address: 10.10.1.10
    volumes:
      - ./shared:/shared    # 挂载共享目录
    command: /bin/bash -c "
      # 安装基础网络工具并保持容器运行
      yum update -y -q && 
      yum install -y -q iproute net-tools iputils && 
      tail -f /dev/null"

  # ========== Worker 节点 (3台) ==========
  worker1:
    image: centos:7
    container_name: cluster-worker1
    hostname: worker1
    stdin_open: true
    tty: true
    restart: unless-stopped
    networks:
      cluster-net:
        ipv4_address: 10.10.1.11
    volumes:
      - ./shared:/shared
    command: /bin/bash -c "
      yum update -y -q && 
      yum install -y -q iproute net-tools iputils && 
      tail -f /dev/null"

  worker2:
    image: centos:7
    container_name: cluster-worker2
    hostname: worker2
    stdin_open: true
    tty: true
    restart: unless-stopped
    networks:
      cluster-net:
        ipv4_address: 10.10.1.12
    volumes:
      - ./shared:/shared
    command: /bin/bash -c "
      yum update -y -q && 
      yum install -y -q iproute net-tools iputils && 
      tail -f /dev/null"

  worker3:
    image: centos:7
    container_name: cluster-worker3
    hostname: worker3
    stdin_open: true
    tty: true
    restart: unless-stopped
    networks:
      cluster-net:
        ipv4_address: 10.10.1.13
    volumes:
      - ./shared:/shared
    command: /bin/bash -c "
      yum update -y -q && 
      yum install -y -q iproute net-tools iputils && 
      tail -f /dev/null"

# ========== 自定义网络 ==========
networks:
  cluster-net:
    driver: bridge
    ipam:
      config:
        - subnet: 10.10.1.0/24
```
