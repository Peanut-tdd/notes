### 运行脚本-deplo.sh文件配置

```
#!/bin/bash
set -e

echo "正在启动 CentOS 4 节点集群..."

# 启动所有服务
docker-compose up -d

# 等待容器初始化
sleep 5

# 创建 /etc/hosts 集群解析 (通过临时执行命令)
echo "配置集群节点互信..."
docker exec cluster-master /bin/bash -c "
  echo '10.10.1.10  master' >> /etc/hosts
  echo '10.10.1.11  worker1' >> /etc/hosts
  echo '10.10.1.12  worker2' >> /etc/hosts
  echo '10.10.1.13  worker3' >> /etc/hosts
"

docker exec cluster-worker1 /bin/bash -c "
  echo '10.10.1.10  master' >> /etc/hosts
  echo '10.10.1.11  worker1' >> /etc/hosts
  echo '10.10.1.12  worker2' >> /etc/hosts
  echo '10.10.1.13  worker3' >> /etc/hosts
"

docker exec cluster-worker2 /bin/bash -c "
  echo '10.10.1.10  master' >> /etc/hosts
  echo '10.10.1.11  worker1' >> /etc/hosts
  echo '10.10.1.12  worker2' >> /etc/hosts
  echo '10.10.1.13  worker3' >> /etc/hosts
"

docker exec cluster-worker3 /bin/bash -c "
  echo '10.10.1.10  master' >> /etc/hosts
  echo '10.10.1.11  worker1' >> /etc/hosts
  echo '10.10.1.12  worker2' >> /etc/hosts
  echo '10.10.1.13  worker3' >> /etc/hosts
"

echo "集群部署完成！"
echo ""
echo "节点信息："
echo "  master   : 10.10.1.10"
echo "  worker1  : 10.10.1.11"
echo "  worker2  : 10.10.1.12"
echo "  worker3  : 10.10.1.13"
echo ""
echo "进入 master 节点：docker exec -it cluster-master /bin/bash"
echo "测试网络：docker exec cluster-master ping -c 2 worker1"
```



给脚本赋执行权限

```
chmod +x deploy.sh
./deploy.sh
```



验证集群通信

```
# 1. 进入 Master 节点
docker exec -it cluster-master /bin/bash

# 2. 在容器内测试（支持主机名解析）
ping worker1
ping worker2
ping worker3

# 3. 测试共享目录
touch /shared/test.txt  # 在 master 创建
ls /shared/              # 在 worker1 查看，文件应存在
```
